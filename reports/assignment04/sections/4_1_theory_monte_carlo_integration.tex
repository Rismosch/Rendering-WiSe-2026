E[F_N] &= E\left[\frac{1}{N}\sum_{i=1}^N \frac{f(X_i)}{p(X_i)}\right] \\
&= \frac{1}{N} \sum_{i=1}^N \int_\Omega \frac{f(x)}{p(x)}p(x)\,d\mu(x) \\
&= \int_\Omega f(x)\,d\mu(x) \\
&= I
\end{align}
